cmake_minimum_required(VERSION 3.22.1)
project(questllm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama-build)
add_library(questllm SHARED questllm.cpp)
target_include_directories(questllm PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/include)
target_link_libraries(questllm PRIVATE llama log)
